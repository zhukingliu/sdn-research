1. 概述与背景

1.1 什么是 SNMP

SNMP (Simple Network Management Protocol) 是 1988 年首次定义的应用层网络管理协议 (RFC 1157)，至今已有近 40 年历史。它基于 管理者-代理 (Manager-Agent) 模型，使用 UDP 作为传输协议，通过 MIB (管理信息库) 树形结构组织和管理设备数据。

尽管 gNMI、NETCONF 等现代协议兴起，SNMP 仍然是网络设备监控的 "最后防线"，在企业网络、电信、IoT 等领域无处不在。

1.2 发展里程碑

时间里程碑
1988RFC 1157: SNMPv1 发布
1996SNMPv2c 发布 (Community-based)，引入 GetBulk 和 Counter64
2002RFC 3411-3418: SNMPv3 完整安全标准 (USM + VACM)
2004-2020Cisco/Juniper/Huawei/HP 全面支持 SNMPv3
2025SNMP 仍是全球网络设备监控最高普及率协议 (>95%)

5. SNMPv3 安全架构详解

SNMPv3 引入了两套互补的安全子系统: USM (User-based Security Model) 处理认证加密，VACM (View-based Access Control Model) 处理授权和访问控制。

5.1 USM 安全级别

安全级别认证加密适用场景
noAuthNoPriv无无测试环境
authNoPrivHMAC-MD5 / SHA无可信内网监控
authPrivHMAC-SHADES / AES (128/192/256)生产环境 / 跨公网

7. Java SNMP4J 开发实践

SNMP4J -- GET / Walk / GETBULK
// Maven: org.snmp4j:snmp4j:3.7.0 import org.snmp4j.*; import org.snmp4j.smi.*; import org.snmp4j.transport.DefaultUdpTransportMapping; import org.snmp4j.mp.*; public class SnmpManager { private Snmp snmp; private Target<?> target; // Init SNMPv2c session public void initV2c(String host, String community) throws Exception { snmp = new Snmp(new DefaultUdpTransportMapping()); snmp.listen(); CommunityTarget ct = new CommunityTarget(); ct.setCommunity(new OctetString(community)); ct.setAddress(GenericAddress.parse("udp:" + host + "/161")); ct.setVersion(SnmpConstants.version2c); ct.setRetries(2); ct.setTimeout(3000); this.target = ct; } // SNMP GET public String get(String oid) throws Exception { PDU pdu = new PDU(); pdu.add(new VariableBinding(new OID(oid))); pdu.setType(PDU.GET); ResponseEvent resp = snmp.send(pdu, target); if (resp.getResponse() != null) return resp.getResponse().get(0).getVariable().toString(); return null; } // SNMP WALK (GETNEXT loop) public java.util.Map<String,String> walk(String baseOid) throws Exception { java.util.Map<String,String> results = new java.util.LinkedHashMap<>(); OID currentOid = new OID(baseOid); while (true) { PDU pdu = new PDU(); pdu.add(new VariableBinding(currentOid)); pdu.setType(PDU.GETNEXT); ResponseEvent ev = snmp.send(pdu, target); if (ev == null || ev.getResponse() == null) break; VariableBinding vb = ev.getResponse().get(0); String next = vb.getOid().toDottedString(); if (!next.startsWith(baseOid)) break; results.put(next, vb.getVariable().toString()); currentOid = vb.getOid(); } return results; } // GETBULK for batch retrieval public void getBulk(String oid) throws Exception { PDU pdu = new PDU(); pdu.add(new VariableBinding(new OID(oid))); pdu.setType(PDU.GETBULK); pdu.setMaxRepetitions(10); pdu.setNonRepeaters(0); ResponseEvent resp = snmp.send(pdu, target); resp.getResponse().getVariableBindings().forEach(System.out::println); } public void close() throws Exception { snmp.close(); } public static void main(String[] args) throws Exception { SnmpManager m = new SnmpManager(); m.initV2c("192.168.1.1", "public"); System.out.println("Uptime: " + m.get(".1.3.6.1.2.1.1.3.0")); m.walk(".1.3.6.1.2.1.2.2.1").forEach( (oid, val) -> System.out.println(oid + " = " + val)); m.close(); } }
